import os, random, sqlite3, time, statistics
from concurrent.futures import ThreadPoolExecutor
DB = "/config/plex/Library/Application Support/Plex Media Server/Plug-in Support/Databases/com.plexapp.plugins.library.db"
c = sqlite3.connect(f"file:{DB}?mode=ro", uri=True, timeout=30)
t = time.time()
rows = c.execute("""SELECT mi.library_section_id, mp.file FROM metadata_items mi JOIN media_items mdi ON mdi.metadata_item_id=mi.id AND mdi.deleted_at IS NULL
JOIN media_parts mp ON mp.media_item_id=mdi.id AND mp.deleted_at IS NULL WHERE mi.library_section_id IN (1,2,12) AND mi.metadata_type IN (1,4) AND mi.deleted_at IS NULL""").fetchall()
print(f"parts={len(rows)} query={time.time()-t:.2f}s", {s: sum(1 for r in rows if r[0]==s) for s in (1,2,12)})
random.seed(7); sample = [r[1] for r in random.sample(rows, 400)]
def timed(p):
    t = time.perf_counter()
    try: os.stat(p)
    except OSError: pass
    return time.perf_counter() - t
for n in (1, 16, 64):
    part = sample[:100] if n == 1 else sample[100:250] if n == 16 else sample[250:]
    t = time.time()
    with ThreadPoolExecutor(n) as ex: lat = list(ex.map(timed, part))
    wall = time.time() - t
    print(f"threads={n}: {len(part)} stats in {wall:.2f}s -> {len(part)/wall:.1f}/s; median {statistics.median(lat)*1000:.1f}ms p95 {sorted(lat)[int(len(lat)*.95)]*1000:.1f}ms")
